--Top 3 products in every category
SELECT Category,
       Product_Name,
       Total_Revenue
FROM (
    SELECT p.Category,
           p.Product_Name,
           SUM(s.Quantity * TO_NUMBER(REPLACE(REPLACE(p.Unit_Price_USD, '$', ''), ',', ''))) AS Total_Revenue,
           RANK() OVER (PARTITION BY p.Category ORDER BY SUM(s.Quantity * TO_NUMBER(REPLACE(REPLACE(p.Unit_Price_USD, '$', ''), ',', ''))) DESC) AS Revenue_Rank
    FROM Sales_table s
    INNER JOIN Products_table p
        ON s.ProductKey = p.ProductKey
    GROUP BY p.Category, p.Product_Name
)
WHERE Revenue_Rank <= 3
ORDER BY Category, Revenue_Rank;

--Top 5 customers in every state 
SELECT State,
       Name AS Customer_Name,
       Total_Revenue
FROM (
    SELECT C.State,
           C.Name,
           SUM(S.Quantity * TO_NUMBER(REPLACE(REPLACE(P.Unit_Price_USD, '$', ''), ',', ''))) AS Total_Revenue,
           RANK() OVER (
               PARTITION BY C.State
               ORDER BY SUM(S.Quantity * TO_NUMBER(REPLACE(REPLACE(P.Unit_Price_USD, '$', ''), ',', ''))) DESC
           ) AS Customer_Revenue_Rank
    FROM Sales_table S
    INNER JOIN Products_table P
        ON S.ProductKey = P.ProductKey
    INNER JOIN Customers_table C
        ON S.CustomerKey = C.CustomerKey
    GROUP BY C.State, C.Name
)
WHERE Customer_Revenue_Rank <= 5
ORDER BY State, Customer_Revenue_Rank;

--Month-over-month revenue growth 
WITH Monthly_Revenue AS (
    SELECT TO_CHAR(s.Order_Date, 'YYYY-MM') AS Month,
           SUM(s.Quantity * TO_NUMBER(REPLACE(REPLACE(p.Unit_Price_USD, '$', ''), ',', ''))) AS Revenue
    FROM Sales_table s
    INNER JOIN Products_table p
        ON s.ProductKey = p.ProductKey
    GROUP BY TO_CHAR(s.Order_Date, 'YYYY-MM')
)
SELECT Month,
       Revenue,
       LAG(Revenue) OVER (ORDER BY Month) AS Prev_Month_Revenue,
       ROUND(((Revenue - LAG(Revenue) OVER (ORDER BY Month)) 
              / LAG(Revenue) OVER (ORDER BY Month)) * 100, 2) AS MoM_Growth_Percent
FROM Monthly_Revenue
ORDER BY Month;

--Find repeat customers 
SELECT C.CustomerKey,
       C.Name AS Customer_Name,
       COUNT(S.Order_Number) AS Total_Orders
FROM Sales_table S
INNER JOIN Customers_table C
    ON S.CustomerKey = C.CustomerKey
GROUP BY C.CustomerKey, C.Name
HAVING COUNT(S.Order_Number) > 1
ORDER BY Total_Orders DESC;

--Find customers who haven't purchased recently. Define a suitable period such as 90 days.
SELECT C.CustomerKey,
       C.Name AS Customer_Name,
       MAX(S.Order_Date) AS Last_Purchase_Date
FROM Sales_table S
INNER JOIN Customers_table C
    ON S.CustomerKey = C.CustomerKey
GROUP BY C.CustomerKey, C.Name
HAVING MAX(S.Order_Date) < SYSDATE - 90
ORDER BY Last_Purchase_Date;

--Customer lifetime value
SELECT Customer_Name,
       Total_Revenue,
       RANK() OVER (ORDER BY Total_Revenue DESC) AS Revenue_Rank
FROM (
    SELECT C.Name AS Customer_Name,
           SUM(S.Quantity * TO_NUMBER(REPLACE(REPLACE(P.Unit_Price_USD, '$', ''), ',', ''))) AS Total_Revenue
    FROM Sales_table S
    INNER JOIN Products_table P
        ON S.ProductKey = P.ProductKey
    INNER JOIN Customers_table C
        ON S.CustomerKey = C.CustomerKey
    GROUP BY C.Name
)
ORDER BY Revenue_Rank;

--Find profitable vs unprofitable products 
WITH Product_Profit AS (
    SELECT P.Product_Name,
           P.Category,
           SUM(S.Quantity * TO_NUMBER(REPLACE(REPLACE(P.Unit_Price_USD, '$', ''), ',', ''))) AS Revenue,
           SUM(S.Quantity * (TO_NUMBER(REPLACE(REPLACE(P.Unit_Price_USD, '$', ''), ',', ''))
                            - TO_NUMBER(REPLACE(REPLACE(P.Unit_Cost_USD, '$', ''), ',', '')))) AS Profit
    FROM Sales_table S
    INNER JOIN Products_table P
        ON S.ProductKey = P.ProductKey
    GROUP BY P.Product_Name, P.Category
)
SELECT Product_Name,
       Category,
       Revenue,
       Profit,
       ROUND((Profit / Revenue) * 100, 2) AS Profit_Percent,
       CASE
         WHEN (Profit / Revenue) * 100 > 20 THEN 'Highly Profitable'
         WHEN (Profit / Revenue) * 100 BETWEEN 10 AND 20 THEN 'Profitable'
         WHEN (Profit / Revenue) * 100 BETWEEN 0 AND 10 THEN 'Low Profit'
         ELSE 'Loss'
       END AS Profit_Category
FROM Product_Profit
ORDER BY Profit_Percent DESC;
